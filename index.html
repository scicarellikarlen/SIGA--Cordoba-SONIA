/** * SIGA - Sistema Integrado de Gestión Atmosférica (Provincia de Córdoba) * Buscador abierto para todas las localidades + Windy dinámico + Resiliencia */ (function () { 'use strict'; // CONFIGURACIÓN INICIAL (Córdoba Capital por defecto) const CONFIG = { lat: -31.4201, lon: -64.1888, locationName: 'Córdoba Capital, Córdoba', POLL_INTERVAL: 5 * 60 * 1000, MAX_RETRIES: 4, BASE_DELAY: 1200, STORAGE_KEY: 'siga_cordoba_weather_cache', OPEN_METEO_FORECAST: 'https://api.open-meteo.com/v1/forecast', GEOCODING_URL: 'https://geocoding-api.open-meteo.com/v1/search', WINDY_BASE: 'https://embed.windy.com/embed2.html' }; const $ = id => document.getElementById(id); function sleep(ms) { return new Promise(resolve => setTimeout(resolve, ms)); } async function fetchWithRetry(url, options = {}, retries = CONFIG.MAX_RETRIES) { let lastError; for (let attempt = 0; attempt <= retries; attempt++) { try { const controller = new AbortController(); const timeout = setTimeout(() => controller.abort(), 12000); const res = await fetch(url, { ...options, signal: controller.signal, headers: { 'Accept': 'application/json', ...(options.headers || {}) } }); clearTimeout(timeout); if (!res.ok) throw new Error(`HTTP ${res.status}`); return await res.json(); } catch (err) { lastError = err; if (attempt === retries) break; const delay = CONFIG.BASE_DELAY * Math.pow(2, attempt) + Math.random() * 400; await sleep(delay); } } throw lastError; } // BÚSQUEDA DE LOCALIDADES EN CÓRDOBA async function buscarLocalidad(query) { const errorDiv = $('search-error'); errorDiv.style.display = 'none'; try { // Forzamos la búsqueda acotada a Córdoba, Argentina para mayor precisión const url = `${CONFIG.GEOCODING_URL}?name=${encodeURIComponent(query + " Córdoba")}&count=5&language=es&format=json`; const data = await fetchWithRetry(url); if (!data.results || data.results.length === 0) { // Segundo intento general por si el usuario tipeó solo el nombre del pueblo/ciudad const urlAlt = `${CONFIG.GEOCODING_URL}?name=${encodeURIComponent(query)}&count=5&language=es&format=json`; const dataAlt = await fetchWithRetry(urlAlt); // Filtrar preferentemente resultados en Argentina/Córdoba si existen const matchAr = dataAlt.results?.find(r => r.country === "Argentina"); if (matchAr) { aplicarNuevaUbicacion(matchAr); return; } errorDiv.textContent = `No se encontró "${query}". Intente con otra localidad de Córdoba.`; errorDiv.style.display = 'block'; return; } // Tomamos el primer resultado relevante aplicarNuevaUbicacion(data.results[0]); } catch (err) { errorDiv.textContent = 'Error al buscar la localidad. Verifique su conexión.'; errorDiv.style.display = 'block'; } } function aplicarNuevaUbicacion(ubicacion) { CONFIG.lat = ubicacion.latitude; CONFIG.lon = ubicacion.longitude; CONFIG.locationName = `${ubicacion.name}, ${ubicacion.admin1 || 'Córdoba'}, Argentina`; // Actualizar datos del clima y recargar el mapa centrado en la nueva localidad updateWeather(true); loadWindyMap(); } // PERSISTENCIA CACHÉ function saveCache(data) { try { const payload = { data, timestamp: Date.now(), location: { lat: CONFIG.lat, lon: CONFIG.lon, name: CONFIG.locationName } }; localStorage.setItem(CONFIG.STORAGE_KEY, JSON.stringify(payload)); } catch (e) {} } function loadCache() { try { const raw = localStorage.getItem(CONFIG.STORAGE_KEY); if (!raw) return null; const parsed = JSON.parse(raw); if (Date.now() - parsed.timestamp < 2 * 60 * 60 * 1000) return parsed; } catch (e) {} return null; } // API METEOROLÓGICA OPEN-METEO async function fetchWeatherData() { const params = new URLSearchParams({ latitude: CONFIG.lat, longitude: CONFIG.lon, current: [ 'temperature_2m', 'relative_humidity_2m', 'apparent_temperature', 'precipitation', 'cloud_cover', 'pressure_msl', 'wind_speed_10m' ].join(','), timezone: 'America/Argentina/Cordoba', wind_speed_unit: 'kmh' }); const json = await fetchWithRetry(`${CONFIG.OPEN_METEO_FORECAST}?${params}`); if (!json.current) throw new Error('Respuesta inválida'); const c = json.current; return { temp: Math.round(c.temperature_2m * 10) / 10, feelsLike: Math.round(c.apparent_temperature * 10) / 10, humidity: Math.round(c.relative_humidity_2m), wind: Math.round(c.wind_speed_10m), pressure: Math.round(c.pressure_msl), precip: Math.round((c.precipitation || 0) * 10) / 10, clouds: Math.round(c.cloud_cover || 0), updatedAt: new Date().toISOString() }; } function renderWeather(data, isOffline = false) { const panel = document.querySelector('.weather-panel'); panel.classList.add('updating'); $('temp').textContent = `${data.temp}°C`; $('feels-like').textContent = `Sensación: ${data.feelsLike}°C`; $('humidity').textContent = `${data.humidity}%`; $('wind').textContent = `${data.wind} km/h`; $('pressure').textContent = `${data.pressure} hPa`; $('precip').textContent = `${data.precip} mm`; $('clouds').textContent = `${data.clouds}%`; $('location-name').textContent = CONFIG.locationName; $('coords').textContent = `${CONFIG.lat.toFixed(2)}, ${CONFIG.lon.toFixed(2)}`; $('last-update').textContent = `Última actualización: ${new Date().toLocaleTimeString('es-AR')}`; const badge = $('offline-badge'); const status = $('connection-status'); if (isOffline) { badge.classList.remove('hidden'); status.textContent = '● Offline (caché)'; status.className = 'status offline'; } else { badge.classList.add('hidden'); status.textContent = '● En línea'; status.className = 'status online'; } setTimeout(() => panel.classList.remove('updating'), 900); } // MAPA WINDY DINÁMICO let mapRetryCount = 0; const MAX_MAP_RETRIES = 3; function buildWindyUrl() { const params = new URLSearchParams({ lat: CONFIG.lat, lon: CONFIG.lon, detailLat: CONFIG.lat, detailLon: CONFIG.lon, zoom: 8, level: 'surface', overlay: 'satellite', product: 'ecmwf', menu: '', message: 'true', marker: 'true', calendar: 'now', type: 'map', location: 'coordinates', detail: 'true', metricWind: 'km/h', metricTemp: '°C' }); return `${CONFIG.WINDY_BASE}?${params.toString()}`; } function loadWindyMap() { const iframe = $('windy-iframe'); const loading = $('map-loading'); loading.classList.remove('hidden'); iframe.src = buildWindyUrl(); iframe.onload = () => { loading.classList.add('hidden'); mapRetryCount = 0; }; iframe.onerror = () => { if (mapRetryCount < MAX_MAP_RETRIES) { mapRetryCount++; setTimeout(loadWindyMap, 1500 * mapRetryCount); } else { loading.textContent = 'No se pudo cargar el mapa.'; } }; } async function updateWeather() { try { const data = await fetchWeatherData(); renderWeather(data, false); saveCache(data); } catch (err) { const cached = loadCache(); if (cached && cached.data) { renderWeather(cached.data, true); } else { renderWeather({ temp: '--', feelsLike: '--', humidity: '--', wind: '--', pressure: '--', precip: '--', clouds: '--' }, true); $('connection-status').textContent = '● Error de conexión'; $('connection-status').className = 'status error'; } } } function init() { const cached = loadCache(); if (cached && cached.data) { renderWeather(cached.data, true); } loadWindyMap(); updateWeather(); setInterval(updateWeather, CONFIG.POLL_INTERVAL); // Eventos del Buscador const searchBtn = $('search-btn'); const searchInput = $('search-input'); searchBtn.addEventListener('click', () => { const query = searchInput.value.trim(); if (query) buscarLocalidad(query); }); searchInput.addEventListener('keypress', (e) => { if (e.key === 'Enter') { const query = searchInput.value.trim(); if (query) buscarLocalidad(query); } }); $('reload-map').addEventListener('click', () => { mapRetryCount = 0; loadWindyMap(); }); } if (document.readyState === 'loading') { document.addEventListener('DOMContentLoaded', init); } else { init(); } })();
