import requests def validar_temperatura(temp_actual, temp_min_hist=-10.0, temp_max_hist=45.0): """ Verifica que la temperatura esté dentro de rangos físicos lógicos y no desentone bruscamente con los límites históricos de la región. """ # 1. Rango absoluto de seguridad para evitar datos infinitos o nulos corruptos if temp_actual < -20.0 or temp_actual > 55.0: return False, "Temperatura fuera de rango absoluto." # 2. Margen de tolerancia respecto al histórico local (ej: +/- 15 grados) margen = 15.0 if temp_actual < (temp_min_hist - margen) or temp_actual > (temp_max_hist + margen): return False, "Temperatura anómala respecto al registro histórico." return True, "OK" def validar_precipitacion(precip_diaria, acumulado_mensual, limite_diario=150.0, limite_mensual=500.0): """ Controla que los milímetros caídos diarios y el acumulado mensual no superen umbrales lógicos o físicos imposibles para la zona. """ if precip_diaria < 0.0 or precip_diaria > limite_diario: return False, "Precipitación diaria fuera de rango lógico." if acumulado_mensual < 0.0 or acumulado_mensual > limite_mensual: return False, "Acumulado mensual excedido (posible error de acumulación del sensor)." return True, "OK" def consultar_fuente_alternativa(lat, lon): """ Realiza una consulta a una API de respaldo (ej. Open-Meteo u otra fuente oficial) cuando la fuente principal falla o arroja datos absurdos. Marcando registro en revisión.") return { "temperatura": temp, "precipitacion": precip_diaria, "acumulado_mensual": acumulado_mes, "estado": "en_revision_manual" } print("Datos primarios validados correctamente.") return datos_primarios
