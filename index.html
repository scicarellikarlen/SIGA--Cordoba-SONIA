import requests def obtener_clima_smn(estacion_id=""): """ Consulta el feed oficial del Servicio Meteorológico Nacional (SMN) para extraer la temperatura actual, humedad y pronóstico sin depender de APIs genéricas. """ # Endpoint público de estaciones del SMN (Argentina) url_smn = "https://www.smn.gob.ar/api/v1/weather" try: headers = { "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) SIGA-App" } response = requests.get(url_smn, headers=headers, timeout=10) if response.status_code == 200: data = response.json() # Si pasás una estación o querés filtrar por tu zona (ej: Córdoba / Sierras Chicas) # Buscamos el nodo correspondiente en la respuesta del SMN for estacion in data: # Aquí podés filtrar por nombre de estación o ID según el JSON del SMN if estacion_id.lower() in estacion.get("name", "").lower(): return { "temperatura": estacion.get("weather", {}).get("temp"), "humedad": estacion.get("weather", {}).get("humidity"), "descripcion": estacion.get("weather", {}).get("description"), "fuente": "Servicio Meteorológico Nacional" } # Si no encuentra una estación específica, devuelve el primer registro o fallback return {"error": "Estación no encontrada en el SMN"} except Exception as e: print(f"Error al conectar con el SMN: {e}") return None
