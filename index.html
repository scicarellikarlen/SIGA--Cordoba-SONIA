// Script para alternar entre el Modo Dior (Manteca / Verde Bosque) y el Modo Tecnológico (Azul / Verde Flúor) document.addEventListener('DOMContentLoaded', () => { const toggleBtn = document.getElementById('dior-mode-btn'); // Asegúrate de que este sea el ID de tu botón en el HTML if (toggleBtn) { toggleBtn.addEventListener('click', (e) => { e.preventDefault(); document.body.classList.toggle('tech-mode'); // Opcional: Cambiar el texto del botón según el modo activo if (document.body.classList.contains('tech-mode')) { toggleBtn.textContent = "Cambiar a Modo Dior"; } else { toggleBtn.textContent = "Cambiar a Modo Tecnológico"; } }); } });
